    def __init__(self):
        self.base_dir=os.path.dirname(os.path.abspath(sys.argv[0]))
        self.bindings_path=self._find_bindings()
        if not self.bindings_path:
            self.bindings_path=os.path.join(self.base_dir,"bindings.ini")
        print("BINDINGS PATH:", self.bindings_path)
        self.ensure_bindings_integrity()

    def _find_bindings(self):
        candidates=[]
        candidates.append(os.path.join(self.base_dir,"bindings.ini"))
        parent=os.path.abspath(os.path.join(self.base_dir,".."))
        candidates.append(os.path.join(parent,"bindings.ini"))
        for entry in os.listdir(self.base_dir):
            full=os.path.join(self.base_dir,entry)
            if os.path.isdir(full):
                candidates.append(os.path.join(full,"bindings.ini"))
                for sub in os.listdir(full):
                    sub_full=os.path.join(full,sub)
                    if os.path.isdir(sub_full):
                        candidates.append(os.path.join(sub_full,"bindings.ini"))
        for path in candidates:
            if os.path.exists(path):
                return path
        return None

    def ensure_bindings_integrity(self):
        if not os.path.exists(self.bindings_path):
            open(self.bindings_path, "w", encoding="utf-8").write(self.DEFAULT_TEMPLATE)
            return

        text = open(self.bindings_path, "r", encoding="utf-8").read()
        orig = text

        if not text.strip():
            open(self.bindings_path, "w", encoding="utf-8").write(self.DEFAULT_TEMPLATE)
            return

        lines = text.splitlines()
        found_default = False
        first_section_index = None

        for i, line in enumerate(lines):
            s = line.strip()
            if not s or s.startswith(";"):
                continue
            if s.startswith("[") and s.endswith("]"):
                first_section_index = i
                break
            if "=" in s:
                found_default = True
                break

        if not found_default:
            fix_lines = self.DEFAULT_FIX.strip().splitlines()
            if first_section_index is not None:
                lines = lines[:first_section_index] + fix_lines + [""] + lines[first_section_index:]
            else:
                lines = fix_lines + [""] + lines

        text = "\n".join(lines)

        if "[END]" not in text:
            text = text.rstrip() + "\n\n[END]\n"

        if text != orig:
            open(self.bindings_path, "w", encoding="utf-8").write(text)

    def _build_block_from_template(self, full_mapping):
        lines = self.DEFAULT_PRESET.strip().splitlines(True)
        out = []

        for line in lines:
            s = line.strip()

            if not s or s.startswith(";"):
                out.append(line)
                continue

            if "=" in s:
                _, action = s.split("=", 1)
                action = action.strip()
                key = full_mapping.get(action, "#")
                out.append(f"{key} = {action}\n")
            else:
                out.append(line)

        return out

    def load(self):
        p=self.bindings_path
        data={}
        has_default_section=False
        started=False
        if not os.path.exists(p):
            return data,False
        with open(p,"r",encoding="utf-8",errors="ignore") as f:
            for line in f:
                s=line.strip()
                if not s or s.startswith(";"):
                    continue
                if s.startswith("["):
                    if started:
                        break
                    else:
                        continue
                if "=" not in s:
                    continue
                k,v=s.split("=",1)
                key=k.strip()
                action=v.strip()
                if key == "#":
                    has_default_section=True
                    started=True
                    continue
                if not key:
                    continue
                data[key]=action
                has_default_section=True
                started=True
        return data,has_default_section

    def save(self, mapping_dict, preset_name=None):
        self.ensure_bindings_integrity()
        self._backup()

        # 🔥 BUILD FULL MAPPING
        full_mapping = {}

        if not mapping_dict:
            # 👉 použij DEFAULT_PRESET jako zdroj
            for line in self.DEFAULT_PRESET.strip().splitlines():
                s = line.strip()
                if not s or s.startswith(";"):
                    continue
                if "=" in s:
                    k, v = s.split("=", 1)
                    full_mapping[v.strip()] = k.strip()
        else:
            for action in ALL_ACTIONS:
                key = None
                for k, v in mapping_dict.items():
                    if v == action:
                        key = k
                        break
                full_mapping[action] = key if key else "#"

        lines = open(self.bindings_path, "r", encoding="utf-8").readlines()

        out = []
        in_target = False
        found_preset = False
        default_block = True

        for line in lines:
            s = line.strip()

            # === SECTION ===
            if s.startswith("[") and s.endswith("]"):
                section_name = s[1:-1].strip()
                default_block = False

                # 🔥 pokud jsme byli v target sekci → dopiš blok před novou sekcí
                if in_target:
                    out.extend(self._build_block_from_template(full_mapping))
                    in_target = False

                if preset_name and section_name == preset_name:
                    in_target = True
                    found_preset = True

                out.append(line)
                continue

            # === DEFAULT BLOCK ===
            if preset_name is None and default_block:
                if "=" in s and not s.startswith(";"):
                    continue

                if s.startswith("; ===== END DEFAULT"):
                    out.extend(self._build_block_from_template(full_mapping))
                    out.append(line)
                    default_block = False
                    continue

                out.append(line)
                continue

            # === PRESET BLOCK ===
            if preset_name is not None and in_target:
                if "=" in s and not s.startswith(";"):
                    continue

                out.append(line)
                continue

            out.append(line)

        # 🔥 pokud soubor skončil a jsme pořád v sekci
        if in_target:
            out.extend(self._build_block_from_template(full_mapping))

        # === CREATE NEW PRESET ===
        if preset_name and not found_preset:
            new_block = [f"\n[{preset_name}]\n"]
            new_block.extend(self._build_block_from_template(full_mapping))

            new_out = []
            inserted = False

            for line in out:
                if not inserted and line.strip() == "[END]":
                    new_out.extend(new_block)
                    inserted = True
                new_out.append(line)

            if not inserted:
                new_out.extend(new_block)
                new_out.append("\n[END]\n")

            out = new_out

        open(self.bindings_path, "w", encoding="utf-8").writelines(out)

    def _backup(self):
        MAX_BACKUPS=15
        backup_dir=os.path.join(self.base_dir,"backups");os.makedirs(backup_dir,exist_ok=True)
        base_name=os.path.basename(self.bindings_path);backups=[]
        for f in os.listdir(backup_dir):
            if f.startswith(base_name+".backup"):
                p=os.path.join(backup_dir,f);backups.append((f,p,os.path.getmtime(p)))
        backups.sort(key=lambda x:x[2])
        while len(backups)>=MAX_BACKUPS:
            oldest=backups.pop(0)
            try: os.remove(oldest[1])
            except Exception: pass
        counter=0
        while True:
            backup_name=f"{base_name}.backup{counter}"
            backup_path=os.path.join(backup_dir,backup_name)
            if not os.path.exists(backup_path): break
            counter+=1
        with open(self.bindings_path,"r",encoding="utf-8") as original, \
            open(backup_path,"w",encoding="utf-8") as backup:
            backup.write(original.read())
        return backup_name

    def load_presets(self):
        presets={}
        current=None
        if not os.path.exists(self.bindings_path):
            return presets
        with open(self.bindings_path,"r",encoding="utf-8",errors="ignore") as f:
            for line in f:
                s=line.strip()
                if not s or s.startswith(";"):
                    continue
                if s.startswith("[") and s.endswith("]"):
                    name=s[1:-1].strip()
                    if name=="END":
                        break
                    current=name
                    presets[current]={}
                    continue
                if "=" in s and current:
                    k,v=s.split("=",1)
                    presets[current][k.strip()]=v.strip()
        return presets

    def parse_section(self, name):
        # [TID PROFILE - NAME]
        m = re.match(r"^([0-9A-F]{8})\s+(.+?)\s*-\s*(.+)$", name)
        if not m:
            return None, None, None
        return m.group(1), m.group(2), m.group(3)

    def find_profiles_for_tid(self, tid):
        presets = self.load_presets()
        profiles = set()

        for name in presets.keys():
            t, p, _ = self.parse_section(name)
            if t == tid:
                profiles.add(p)

        return profiles

    def make_unique_profile(self, tid, base_profile):
        profiles = self.find_profiles_for_tid(tid)

        if base_profile not in profiles:
            return base_profile

        i = 2
        while True:
            p = f"{base_profile}_{i}"
            if p not in profiles:
                return p
            i += 1

        elif vk in (106,107,109,111):key_name={106:"Num*",107:"Num+",109:"Num-",111:"Num/"}[vk]

            key_name="LAlt" if sc in (56,) else "RAlt"

            key_name="RCtrl"

        elif vk==160 or qt==Qt.Key_Shift:
            key_name="LShift" if sc in (42,) else "RShift"
        elif vk==161:
            key_name="RShift"

        elif sc in SC_ENG_SYMBOLS:key_name=SC_ENG_SYMBOLS[sc]
        elif 112<=vk<=123:key_name=f"F{vk-111}"
        elif vk in VK_SPECIAL:key_name=VK_SPECIAL[vk]
        elif vk==32:key_name="Space"
        else:
            key_name=vk_to_english_key_name(vk)
            if not key_name:return

        key_name=str(key_name).strip()
        if not key_name:return
        if len(key_name)==1:key_name=key_name.upper()

        self.mapped_key=key_name
        print("KEY:",vk,sc,qt,"->",key_name)
        QTimer.singleShot(0,self._apply_mapping)

    def mousePressEvent(self, event):
        pos = event.pos()
        for btn in (self.reset_btn, self.close_button):
            if btn.geometry().contains(pos):
                event.ignore()
                return
        btn = event.button()
        if btn == Qt.LeftButton:
            key_name = "LClick"
        elif btn == Qt.RightButton:
            key_name = "RClick"
        elif btn == Qt.MiddleButton:
            key_name = "MClick"
        elif btn == Qt.XButton1:
            key_name = "Mouse4"
        elif btn == Qt.XButton2:
            key_name = "Mouse5"
        else:
            return

        self.mapped_key = key_name
        QTimer.singleShot(0, self._apply_mapping)
        
    def wheelEvent(self,event):
        child = self.childAt(event.pos())
        if isinstance(child, QPushButton):
            event.ignore()
            return

        key_name="MWheelUp" if event.angleDelta().y()>0 else "MWheelDown"
        self.mapped_key=key_name
        QTimer.singleShot(0,self._apply_mapping)

    def closeEvent(self,event):
        user32.ActivateKeyboardLayout(self._original_layout,0)
        super().closeEvent(event)
        
    def _show_duplicate_warning(self,key_name):
        self.label.setText(f"<p style='color:#FFFFFF;text-align:center;padding:10px;'>This key <span style='color:#FF0000;'> {key_name} </span>is already mapped.<br>Choose another.</p>")
        parent=self.parent()
        if parent and hasattr(parent,"show_status_message"):
            parent.show_status_message(f" Key '{key_name}' already mapped.")
        self.key_mapped_signal.emit()

    def _start_close_timer(self): QTimer.singleShot(1000,self.close)

    def check_gamepad(self):
        buttons = get_gamepad_buttons()
        if buttons is None:
            return

        if buttons & XINPUT_GUIDE:
            self._apply_gamepad_mapping("Guide")

    def _apply_gamepad_mapping(self, action):
        if getattr(self,"_mapping_applied",False):
            return

        parent=self.parent()
        if not parent:
            return

        self._mapping_applied=True

        if hasattr(self,"pad_timer"):
            self.pad_timer.stop()

        target=None
        for dk,ck in parent.key_mappings.items():
            if parent.buttons[dk] is self.button:
                target=GAMEPAD_MAP.get(ck)
                break

        if not target:
            return

        # 🔥 stejné chování jako klávesy
        for btn in parent.buttons.values():
            if btn is self.button:
                continue
            existing=btn.property("mapped_key")
            if existing and existing!="#" and canonical(existing)==canonical(action):
                self._mapping_applied=False
                self._show_duplicate_warning(action)
                return

        parent.update_label(self.button, action)
        self.button.setProperty("mapped_key", action)

        # 🔥 KLÍČOVÉ – uložit přes save_config (ne ručně!)
        parent.save_config()

        self.label.setText(f"<p style='text-align:center;'>Mapped:<br><span style='color:#4eed4e;font-weight:700;'>{action}</span></p>")

        if hasattr(parent,"show_status_message"):
            parent.show_status_message(f" Mapped: {action}")

        self.key_mapped_signal.emit()
    
    @pyqtSlot()
    def _apply_mapping(self):
        if getattr(self,"_mapping_applied",False):
            return
        parent=self.parent()
        key_name=getattr(self,"mapped_key",None)
        if not parent or not key_name:
            return
        self._mapping_applied=True
        if hasattr(self,"pad_timer"):
            self.pad_timer.stop()
        key_name=str(key_name).strip()
        if not key_name or key_name=="#":
            self._mapping_applied=False
            return
        if len(key_name)==1:
            key_name=key_name.upper()
        target=None
        for dk,ck in parent.key_mappings.items():
            if parent.buttons[dk] is self.button:
                target=GAMEPAD_MAP.get(ck)
                break
        if not target:
            return
        for btn in parent.buttons.values():
            if btn is self.button:
                continue
            existing=btn.property("mapped_key")
            if existing and existing!="#" and canonical(existing)==canonical(key_name):
                self._mapping_applied=False
                self._show_duplicate_warning(key_name.upper())
                return
        if self.is_emulator_reserved(key_name):
            warn=EmulatorKeyWarningDialog(self)
            if warn.exec_()!=QDialog.Accepted:
                self._mapping_applied=False
                return
        parent.update_label(self.button,key_name)
        self.button.setProperty("mapped_key",key_name)
        parent.save_config()
        self.label.setText(f"<p style='text-align:center;font-family:arial;font-size:15px;'>Mapped key:<br><span style='color:#4eed4e;font-weight:700;'>{key_name.upper()}</span></p>")
        if hasattr(parent,"show_status_message"):
            parent.show_status_message(f" Key mapped: {key_name.upper()}")
        print("DEBUG KEY:",key_name)
        self.key_mapped_signal.emit()
        
    def is_emulator_reserved(self, key_name):
        return canonical(key_name) in EMULATOR_RESERVED_KEYS

    def _reset_mapping(self):
        parent=self.parent()
        if not parent:return
        target=None
        for dk,ck in parent.key_mappings.items():
            if parent.buttons[dk] is self.button:
                target=GAMEPAD_MAP.get(ck)
                break
        if not target:return
        mapping=parent.current_bindings.copy()
        for k,v in list(mapping.items()):
            if canonical(v)==canonical(target):
                del mapping[k]
        parent.binding_manager.save(mapping,parent.active_preset)
        parent.current_bindings=mapping
        parent.hid_winkey=build_reverse_map(mapping)
        self.button.setProperty("mapped_key",None)
        parent.update_label(self.button,None)
        parent.show_status_message(" Mapping removed")
        self.label.setText("<p style='text-align:center;font-size:14px;'>Mapping removed</p>")
        self.key_mapped_signal.emit()

import json

class GameDatabase:
    def __init__(self):
        base_dir = os.path.dirname(os.path.abspath(sys.argv[0]))
        self.db_path = os.path.join(base_dir, "database", "database.json")
        self.data = {}
        self._load()

    def _load(self):
        if not os.path.exists(self.db_path):
            print("DB: database.json not found")
            self.data = {}
            return

        try:
            with open(self.db_path, "r", encoding="utf-8") as f:
                self.data = json.load(f)
        except Exception as e:
            print("DB LOAD ERROR:", e)
            self.data = {}

    def get_name(self, tid):
        if not tid:
            return None
        tid = tid.upper()
        entry = self.data.get(tid)
        if not entry:
            return None

        # očekáváme formát: { "4D5307E6": { "name": "Forza Horizon" } }
        if isinstance(entry, dict):
            return entry.get("name")

        # fallback kdyby to bylo jen string
        if isinstance(entry, str):
            return entry

        return None

class TooltipFilter(QObject):
    def __init__(self,tooltip,text):
        super().__init__();self.tooltip=tooltip;self.text=text

    def eventFilter(self,obj,event):
        if not self.tooltip: return False
        t=event.type()
        if t==QEvent.Enter:
            self.tooltip.show_text(self.text,QCursor.pos()+QPoint(14,16))
        elif t==QEvent.Leave:
            self.tooltip.hide()
        return False

    def save_config(self):
        mapping={}
        used_keys=set()
        used_actions=set()
        for dk,ck in self.key_mappings.items():
            key=self.buttons[dk].property("mapped_key")
            if not key:
                continue
            key=str(key).strip()
            if not key or key=="#":
                continue
            if len(key)==1:
                key=key.upper()
            ckey=canonical(key)

            action=GAMEPAD_MAP.get(ck)

    def extract_title_id_from_file(self, path):
        try:
            with open(path, "rb") as f:
                data = f.read()

            if data[0:4] != b"XEX2":
                print("Not XEX")
                return None

            # offset na optional headers
            header_offset = int.from_bytes(data[8:12], "big")
            header_count  = int.from_bytes(data[12:16], "big")

            print("HEADER COUNT:", header_count)

            for i in range(header_count):
                entry_offset = 16 + i * 8

                header_id   = int.from_bytes(data[entry_offset:entry_offset+4], "big")
                header_data = int.from_bytes(data[entry_offset+4:entry_offset+8], "big")

                # 🔥 execution info header
                if header_id == 0x00040006:
                    info_offset = header_data

                    # TitleID je uvnitř struktury
                    title_id_bytes = data[info_offset+4:info_offset+8]
                    tid = title_id_bytes.hex().upper()

                    print("REAL TID:", tid)
                    return tid

        except Exception as e:
            print("XEX PARSE ERROR:", e)

        return None

    def extract_title_id_from_xex(self, path):
        return self.extract_title_id_from_file(path)
    
    def auto_detect_preset(self, path):
        print("=== AUTO DETECT START ===")
        print("PATH:", path)

        tid = None

        # 🔥 1) DB match podle názvu (NORMALIZOVANÉ – bez mezer/znaků)
        filename = os.path.basename(path).lower()
        filename_clean = re.sub(r"[^a-z0-9]", "", filename)

        print("FILENAME:", filename)
        print("FILENAME CLEAN:", filename_clean)
        print("DB SIZE:", len(self.db.data) if hasattr(self, "db") else "NO DB")

        if hasattr(self, "db") and self.db.data:
            for db_tid, entry in self.db.data.items():

                if isinstance(entry, dict):
                    game_name = entry.get("name", "")
                else:
                    game_name = entry

                game_name_clean = re.sub(r"[^a-z0-9]", "", game_name.lower())

                if game_name_clean and game_name_clean in filename_clean:
                    tid = db_tid.upper()
                    print("TID FROM DB NAME:", tid)
                    break

        # 🔥 2) FOLDER
        if not tid:
            folder = os.path.basename(os.path.dirname(path)).upper()
            if re.fullmatch(r"[0-9A-F]{8}", folder):
                tid = folder
                print("TID FROM FOLDER:", tid)

        # 🔥 3) FILENAME HEX
        if not tid:
            name = os.path.basename(path).upper()
            match = re.search(r"[0-9A-F]{8}", name)
            if match:
                tid = match.group(0)
                print("TID FROM FILENAME:", tid)

        # 🔥 4) XEX SCAN (POSLEDNÍ MOŽNOST)
        if not tid:
            tid = self.extract_title_id_from_file(path)
            print("TID FROM XEX:", tid)

        print("FINAL TID:", tid)

        if not tid:
            self.show_status_message(" No Title ID detected", 3000)
            return

        tid = tid.upper()

        print("PRESETS:")
        for p in self.presets:
            print("->", p)

        # 🔥 MATCH EXISTING (PŘESNÉ, žádné startswith)
        for name in self.presets:
            section = name.strip().upper()
            section_tid = section.split(" ")[0]

            print("CHECK:", section_tid, "vs", tid)

            if section_tid == tid:
                self.preset_combo.setCurrentText(name)
                self.show_status_message(f" Detected preset ({tid})", 3000)
                return

        # 🔥 CREATE NEW
        game_name = self.db.get_name(tid)

        if not game_name:
            game_name = self.extract_game_name(path)

        bm = self.binding_manager
        existing = bm.find_profiles_for_tid(tid)

        if "Default" in existing:
            dlg = DuplicatePresetDialog("Default", self)
            if dlg.exec_() != QDialog.Accepted:
                self.show_status_message(" Cancelled.", 2000)
                return

            profile = bm.make_unique_profile(tid, "Default")
            self.show_status_message(f" Duplicate created: {profile}", 3000)
        else:
            profile = "Default"

        preset_name = f"{tid} {profile} - {game_name}"

        self.binding_manager.save({}, preset_name)
        self.presets = self.binding_manager.load_presets()
        self.preset_combo.addItem(preset_name)
        self.preset_combo.setCurrentText(preset_name)

        self.show_status_message(f" Created preset: {preset_name}", 3000)

    def extract_game_name(self,path,data=None):
        folder=os.path.basename(os.path.dirname(path))
        if len(folder)>3 and not re.fullmatch(r"[0-9A-Fa-f]+",folder):
            return folder

        name=os.path.splitext(os.path.basename(path))[0]
        if len(name)>3 and name.lower()!="default" and not re.fullmatch(r"[0-9A-Fa-f]+",name):
            return name

        if data:
            try:
                strings=re.findall(rb"[ -~]{6,}",data)
                for s in strings:
                    txt=s.decode(errors="ignore").strip()
                    if re.fullmatch(r"[0-9A-Fa-f]+",txt):
                        continue
                    if not re.search(r"[A-Za-z]{3,}",txt):
                        continue
                    if len(txt)>40:
                        continue
                    if not re.match(r"^[A-Za-z0-9 .:'\-]+$",txt):
                        continue
                    return txt
            except:
                pass

        return "Unknown Game"

    def create_or_duplicate_preset(self, tid, profile, game_name):
        bm = self.binding_manager

        existing = bm.find_profiles_for_tid(tid)

        if profile in existing:
            dlg = DuplicatePresetDialog(profile, self)
            if dlg.exec_() != QDialog.Accepted:
                self.show_status_message(" Cancelled.", 2000)
                return None

            new_profile = bm.make_unique_profile(tid, profile)
            self.show_status_message(f" Duplicate created: {new_profile}", 3000)
        else:
            new_profile = profile

        preset_name = f"{tid} {new_profile} - {game_name}"
        return preset_name
