    // Wrapper mode: htmlmixed tokenizes the HTML as usual. When it reaches a
    // quoted style="..." or on*="..." attribute, we split that one HTML string
    // token and pass its contents through CodeMirror's actual CSS/JavaScript mode.
    function defineSktHtmlMode() {
        if (CodeMirror.modes && CodeMirror.modes['skt-html']) return;

        CodeMirror.defineMode('skt-html', function (config) {
            const htmlMode = CodeMirror.getMode(config, 'htmlmixed');
            const cssMode = CodeMirror.getMode(config, 'css');
            const jsMode = CodeMirror.getMode(config, 'javascript');

            function attributeKind(name) {
                const lower = (name || '').toLowerCase();
                if (lower === 'style') return 'css';
                if (/^on[a-z0-9_:-]+$/i.test(lower)) return 'js';
                return null;
            }

            function startState() {
                return {
                    html: htmlMode.startState(),
                    lastAttribute: '',
                    waitingForValue: false,
                    inline: null
                };
            }

            function copyState(state) {
                let inline = null;
                if (state.inline) {
                    inline = {
                        kind: state.inline.kind,
                        mode: state.inline.mode,
                        state: CodeMirror.copyState(state.inline.mode, state.inline.state),
                        contentEnd: state.inline.contentEnd
                    };
                }
                return {
                    html: CodeMirror.copyState(htmlMode, state.html),
                    lastAttribute: state.lastAttribute,
                    waitingForValue: state.waitingForValue,
                    inline: inline
                };
            }

            function tokenInline(stream, state) {
                const inline = state.inline;
                if (!inline) return null;

                if (stream.pos < inline.contentEnd) {
                    const fragment = stream.string.slice(stream.pos, inline.contentEnd);
                    const innerStream = new CodeMirror.StringStream(fragment, config.tabSize || 4);
                    const innerStyle = inline.mode.token(innerStream, inline.state);
                    const consumed = Math.max(1, innerStream.pos);
                    const tokenText = fragment.slice(0, consumed);
                    const isHexColor = inline.kind === 'css' &&
                        /^#(?:[0-9a-f]{3,4}|[0-9a-f]{6}|[0-9a-f]{8})$/i.test(tokenText);
                    stream.pos += Math.min(consumed, inline.contentEnd - stream.pos);
                    return 'skt-inline-' + inline.kind +
                        (innerStyle ? ' ' + innerStyle : '') +
                        (isHexColor ? ' skt-hex-color skt-hex-' + tokenText.slice(1).toLowerCase() : '');
                }

                // The closing quote is kept in the ordinary HTML string color.
                if (stream.pos === inline.contentEnd && !stream.eol()) {
                    stream.next();
                    state.inline = null;
                    return 'string';
                }

                // Defensive recovery for malformed or unexpectedly split values.
                state.inline = null;
                if (!stream.eol()) stream.next();
                return 'string';
            }

            return {
                startState: startState,
                copyState: copyState,
                token: function (stream, state) {
                    // Finish re-tokenizing an inline value before asking htmlmixed
                    // for its next token. htmlmixed has already advanced its own
                    // parser state past the original quoted attribute.
                    if (state.inline) return tokenInline(stream, state);

                    const start = stream.pos;
                    const htmlStyle = htmlMode.token(stream, state.html);
                    const end = stream.pos;
                    const raw = stream.string.slice(start, end);
                    const styleName = String(htmlStyle || '');

                    // XML/HTML mode marks attribute names with the "attribute" token.
                    if (styleName.indexOf('attribute') >= 0) {
                        state.lastAttribute = raw;
                        state.waitingForValue = true;
                        return htmlStyle;
                    }

                    // Whitespace and the equals sign may be separate tokens between
                    // the attribute name and its quoted value.
                    if (state.waitingForValue && /^(?:\s|=)*$/.test(raw)) return htmlStyle;

                    if (state.waitingForValue && styleName.indexOf('string') >= 0) {
                        const kind = attributeKind(state.lastAttribute);
                        const quote = raw.charAt(0);
                        const closesOnThisToken = raw.length >= 2 &&
                            (quote === '"' || quote === "'") && raw.charAt(raw.length - 1) === quote;

                        state.waitingForValue = false;
                        state.lastAttribute = '';

                        // Only handle complete, quoted attribute tokens. Malformed or
                        // multiline values are left to htmlmixed rather than risking
                        // desynchronization of SKT's editor.
                        if (kind && closesOnThisToken) {
                            const mode = kind === 'css' ? cssMode : jsMode;
                            state.inline = {
                                kind: kind,
                                mode: mode,
                                state: mode.startState(),
                                contentEnd: end - 1
                            };
                            // Emit the opening quote as a normal HTML string token;
                            // subsequent calls emit CSS/JS tokens up to the closing quote.
                            stream.pos = start + 1;
                            return htmlStyle;
                        }

                        return htmlStyle;
                    }

                    // Preserve the candidate across harmless separators only. Any
                    // other HTML token means the pending attribute value was absent.
                    if (state.waitingForValue) {
                        state.waitingForValue = false;
                        state.lastAttribute = '';
                    }

                    return htmlStyle;
                }
            };
        });
    }

        function clamp(value, min, max) { return Math.max(min, Math.min(max, value)); }
        function rgbToHsv(r, g, b) {
            r /= 255; g /= 255; b /= 255;
            const max = Math.max(r, g, b), min = Math.min(r, g, b), delta = max - min;
            let h = 0;
            if (delta) {
                if (max === r) h = 60 * (((g - b) / delta) % 6);
                else if (max === g) h = 60 * ((b - r) / delta + 2);
                else h = 60 * ((r - g) / delta + 4);
            }
            if (h < 0) h += 360;
            return { h: h, s: max ? delta / max : 0, v: max };
        }
        function hsvToHex(h, s, v) {
            const c = v * s, x = c * (1 - Math.abs((h / 60) % 2 - 1)), m = v - c;
            let rgb;
            if (h < 60) rgb = [c, x, 0];
            else if (h < 120) rgb = [x, c, 0];
            else if (h < 180) rgb = [0, c, x];
            else if (h < 240) rgb = [0, x, c];
            else if (h < 300) rgb = [x, 0, c];
            else rgb = [c, 0, x];
            return '#' + rgb.map(function (n) { return Math.round((n + m) * 255).toString(16).padStart(2, '0'); }).join('');
        }
        function normalizeHex(value) {
            const raw = String(value || '').trim().replace(/^#/, '');
            if (/^[0-9a-f]{3}$/i.test(raw)) return '#' + raw.split('').map(ch => ch + ch).join('').toLowerCase();
            if (/^[0-9a-f]{4}$/i.test(raw)) return '#' + raw.slice(0, 3).split('').map(ch => ch + ch).join('').toLowerCase();
            if (/^[0-9a-f]{6}$/i.test(raw)) return '#' + raw.toLowerCase();
            if (/^[0-9a-f]{8}$/i.test(raw)) return '#' + raw.slice(0, 6).toLowerCase();
            return null;
        }

            if (original.length === 9) return base + original.slice(7, 9);
            return base;
        }

                cursor.style.left = (hsv.s * 100) + '%';
                cursor.style.top = ((1 - hsv.v) * 100) + '%';
            }
            function updateFromSpectrum(event) {
                const rect = sv.getBoundingClientRect();
                hsv.s = clamp((event.clientX - rect.left) / rect.width, 0, 1);
                hsv.v = 1 - clamp((event.clientY - rect.top) / rect.height, 0, 1);
                selectedHex = hsvToHex(hsv.h, hsv.s, hsv.v);
                hexInput.value = selectedHex;
                preview.style.backgroundColor = selectedHex;
                updateSpectrum();
            }
            sv.addEventListener('pointerdown', function (event) {
                event.preventDefault();
                sv.setPointerCapture(event.pointerId);
                updateFromSpectrum(event);
            });
            sv.addEventListener('pointermove', function (event) {
                if (event.buttons) updateFromSpectrum(event);
            });
            hueInput.addEventListener('input', function () {
                hsv.h = Number(hueInput.value);
                updateSpectrum();
                selectedHex = hsvToHex(hsv.h, hsv.s, hsv.v);
                hexInput.value = selectedHex;
                preview.style.backgroundColor = selectedHex;
            });
            hexInput.addEventListener('input', function () {
                const normalized = normalizeHex(hexInput.value);
                if (normalized) setHex(normalized, true);
            });
            paletteColors.forEach(function (color) {
                const chip = document.createElement('button');
                chip.type = 'button';
                chip.className = 'skt-picker-chip';
                chip.style.backgroundColor = color;

                chip.setAttribute('aria-label', color);

                palette.appendChild(chip);

            panel.querySelector('.skt-picker-cancel').addEventListener('click', closePicker);
            panel.querySelector('.skt-picker-apply').addEventListener('click', function () {
                const normalized = normalizeHex(hexInput.value);
                if (!normalized) { hexInput.focus(); return; }
                const replacement = colorForOriginal(normalized, original);
                cm.replaceRange(replacement, rangeFrom, rangeTo, '+skt-color-picker');
                cm.save();
                closePicker();
                cm.focus();
            });
            setHex(originalHex, true);
            window.setTimeout(function () { hexInput.focus(); hexInput.select(); }, 0);
            document.addEventListener('mousedown', outsidePickerMouseDown, true);
            document.addEventListener('keydown', pickerEscape, true);
        }

        function handleHexSwatchMouseDown(event) {
            if (event.button !== 0) return;
            const target = event.target instanceof Element ? event.target.closest('.cm-skt-hex-color') : null;
            if (!target || !cm.getWrapperElement().contains(target)) return;
            const rect = target.getBoundingClientRect();
            const before = window.getComputedStyle(target, '::before');
            const beforeWidth = parseFloat(before.width) || 0;
            const marginLeft = parseFloat(before.marginLeft) || 0;
            const marginRight = parseFloat(before.marginRight) || 0;
            const swatchRight = rect.left + marginLeft + beforeWidth + marginRight + 2;
            if (event.clientX > swatchRight) return;

            const tokenText = (target.textContent || '').trim();
            if (!/^#(?:[0-9a-f]{3,4}|[0-9a-f]{6}|[0-9a-f]{8})$/i.test(tokenText)) return;
            const linePos = cm.coordsChar({ left: swatchRight + 1, top: rect.top + rect.height / 2 }, 'window');
            const lineText = cm.getLine(linePos.line) || '';
            const matches = Array.from(lineText.matchAll(/#[0-9a-f]{3,4}(?![0-9a-f])|#[0-9a-f]{6}(?![0-9a-f])|#[0-9a-f]{8}(?![0-9a-f])/gi));
            const candidates = matches.filter(match => match[0].toLowerCase() === tokenText.toLowerCase());
            if (!candidates.length) return;
            let chosen = candidates[0];
            let bestDistance = Infinity;
            candidates.forEach(function (match) {
                const distance = Math.abs(match.index - linePos.ch);
                if (distance < bestDistance) { bestDistance = distance; chosen = match; }
            });
            const from = { line: linePos.line, ch: chosen.index };
            const to = { line: linePos.line, ch: chosen.index + chosen[0].length };
            event.preventDefault();
            event.stopPropagation();
            if (typeof event.stopImmediatePropagation === 'function') event.stopImmediatePropagation();
            openPicker(target, from, to, tokenText);
        }
        cm.getWrapperElement().addEventListener('mousedown', handleHexSwatchMouseDown, true);

        const form = textarea.form || document;

        function copySelectionToOriginal() {
            cm.save();
            const anchor = cm.getCursor('anchor');
            const head = cm.getCursor('head');
            const start = cm.indexFromPos(anchor);
            const end = cm.indexFromPos(head);
            try { textarea.focus({ preventScroll: true }); }
            catch (_) { textarea.focus(); }
            try { textarea.setSelectionRange(start, end); } catch (_) { /* ignore */ }
        }

        function copyOriginalBackToEditor(restoreFocus) {
            window.setTimeout(function () {
                const value = textarea.value;
                if (value !== cm.getValue()) cm.setValue(value);
                if (typeof textarea.selectionStart === 'number' && typeof textarea.selectionEnd === 'number') {
                    const start = Math.min(textarea.selectionStart, cm.getValue().length);
                    const end = Math.min(textarea.selectionEnd, cm.getValue().length);
                    cm.setSelection(cm.posFromIndex(start), cm.posFromIndex(end));
                }
                cm.save();
                if (restoreFocus) cm.focus();
            }, 0);
        }

        function isEditorToolbarAction(target) {
            if (!(target instanceof Element)) return false;
            if (target.closest('.CodeMirror')) return false;
            return !!target.closest('button, select, option, a, img[onclick], input[type="button"], input[type="submit"], input[type="reset"], [onclick]');
        }

        form.addEventListener('mousedown', function (event) {
            if (isEditorToolbarAction(event.target)) copySelectionToOriginal();
        }, true);

        form.addEventListener('click', function (event) {
            if (!isEditorToolbarAction(event.target)) return;
            const target = event.target instanceof Element
                ? event.target.closest('button, input[type="button"], input[type="submit"], input[type="reset"]')
                : null;
            const isSubmit = !!(target && (
                target.type === 'submit' ||
                target.value === 'Potwierdź' ||
                target.value === 'Submit'
            ));
            copyOriginalBackToEditor(!isSubmit);
        }, true);

        form.addEventListener('change', function (event) {
            if (isEditorToolbarAction(event.target)) copyOriginalBackToEditor(true);
        }, true);

        console.info('[SKT Code Editor] Aktivní – HTML + inline CSS + event JavaScript.');
        return true;
    }

    function isPrivateMessageEditPage() {
        const query = new URLSearchParams(window.location.search);
        return query.get('do') === 'pm' && query.get('action') === 'edit';
    }

    let scheduled = false;
    let lastUrl = window.location.href;

    function scheduleInitialize() {
        if (scheduled) return;
        scheduled = true;
        window.setTimeout(function () {
            scheduled = false;
            if (!isPrivateMessageEditPage()) return;
            initialize();
        }, 150);
    }

    const observer = new MutationObserver(scheduleInitialize);
    if (document.documentElement) observer.observe(document.documentElement, { childList: true, subtree: true });

    window.addEventListener('popstate', scheduleInitialize, false);
    window.addEventListener('hashchange', scheduleInitialize, false);
    window.setInterval(function () {
        if (window.location.href !== lastUrl) {
            lastUrl = window.location.href;
            scheduleInitialize();
        } else if (isPrivateMessageEditPage()) {
            initialize();
        }
    }, 1000);

    scheduleInitialize();
    window.__sktCodeEditorController = true;
})();
